SELECT PLANT$.Plant, COUNT(DEVICE$.Device) FROM PLANT$
inner join DEVICE$ on DEVICE$.IdPlant_FK = PLANT$.IdPlant
GROUP BY PLANT$.Plant
ORDER BY PLANT$.Plant

SELECT DETTAGLIDEVICE$.[Controller HubTemp], DETTAGLIDEVICE$.[Controller Top Temp], DETTAGLIDEVICE$.[Ambient Temp], DETTAGLIDEVICE$.[Spinner Temp] FROM DETTAGLIDEVICE$
INNER JOIN DEVICE$ ON DEVICE$.IdDevice = DETTAGLIDEVICE$.IdDeviceFK
WHERE DEVICE$.Device = @Device

SELECT DETTAGLIDEVICE$.[Hydraulic Oil Temp], DETTAGLIDEVICE$.[Hydraulic Oil Pressure] FROM DETTAGLIDEVICE$
INNER JOIN DEVICE$ ON DEVICE$.IdDevice = DETTAGLIDEVICE$.IdDeviceFK
WHERE DEVICE$.Device = @Device

SELECT AVG(DETTAGLIDEVICE$.[Nacelle Dir]), AVG(DETTAGLIDEVICE$.[Wind Dir]) FROM DETTAGLIDEVICE$
INNER JOIN DEVICE$ ON DEVICE$.IdDevice = DETTAGLIDEVICE$.IdDeviceFK
WHERE DEVICE$.Device = @Device
GROUP BY CONVERT(DATE,DETTAGLIDEVICE$.Date)
